    /**
     * Bulk accept pending applications. Non-pending rows are silently skipped.
     */
    public function bulkAccept(Request $request): RedirectResponse
    {
        $this->authorize('update', Application::class);

        $ids = $request->validate(['ids' => 'required|array', 'ids.*' => 'integer'])['ids'];

        Application::whereIn('id', $ids)
            ->where('status', 'pending')
            ->update([
                'status' => 'accepted',
                'processed_by' => auth()->id(),
                'processed_at' => now(),
            ]);

        return back()->with('success', 'Selected applications accepted.');
    }

    /**
     * Bulk dismiss pending applications. Non-pending rows are silently skipped.
     */
    public function bulkDismiss(Request $request): RedirectResponse
    {
        $this->authorize('update', Application::class);

        $ids = $request->validate(['ids' => 'required|array', 'ids.*' => 'integer'])['ids'];

        Application::whereIn('id', $ids)
            ->where('status', 'pending')
            ->update([
                'status' => 'dismissed',
                'processed_by' => auth()->id(),
                'processed_at' => now(),
            ]);

        return back()->with('success', 'Selected applications dismissed.');
    }

    /**
     * Re-open a dismissed application back to pending.
     * Gated: application window of the linked academic year must still be open.
     */
    public function reopen(Application $application): RedirectResponse
    {
        $this->authorize('update', $application);

        if (! $application->season?->isApplicationWindowOpen()) {
            return back()->withErrors(['error' => 'The application window is closed.']);
        }

        $application->update(['status' => 'pending']);

        return back()->with('success', 'Application re-opened.');
    }

    /**
     * Hard-delete an application (any status). Requires admin or super_admin.
     */
    public function destroy(Application $application): RedirectResponse
    {
        $this->authorize('delete', $application);

        $application->delete();

        return redirect('/applications')->with('success', 'Application deleted.');
    }
```

> **Note:** The `reopen()` method uses `$application->season`. After Phase 3A Batch 3 is merged and stable, rename this to `$application->academicYear`.

  let selectedIds = $state([]);

  function toggleSelect(id) {
    selectedIds = selectedIds.includes(id) ? selectedIds.filter((x) => x !== id) : [...selectedIds, id];
  }

  function toggleSelectAll(apps) {
    const pending = apps.filter((a) => a.status === 'pending').map((a) => a.id);
    selectedIds = selectedIds.length === pending.length && pending.every((id) => selectedIds.includes(id))
      ? []
      : pending;
  }

  function doBulkAccept() {
    if (selectedIds.length === 0) return;
    router.post('/applications/bulk-accept', { ids: selectedIds }, { onSuccess: () => { selectedIds = []; } });
  }

  function doBulkDismiss() {
    if (selectedIds.length === 0) return;
    router.post('/applications/bulk-dismiss', { ids: selectedIds }, { onSuccess: () => { selectedIds = []; } });
  }

  function doReopen(id) {
    router.put(`/applications/${id}/reopen`, {}, { preserveScroll: true });
  }

  function doDelete(id) {
    if (!confirm('Delete this application? This cannot be undone.')) return;
    router.delete(`/applications/${id}`);
  }
```

    /**
     * Bulk update attendance or submission status for a list of applicants in a session.
     * Status 'present'/'absent' updates attendance_status.
     * Status 'submitted' updates submission_status.
     */
    public function bulkAttendance(Request $request, ExamSession $exam_session): RedirectResponse
    {
        $data = $request->validate([
            'applicant_ids'   => ['required', 'array'],
            'applicant_ids.*' => ['integer'],
            'status'          => ['required', Rule::in(['present', 'absent', 'submitted'])],
        ]);

        $column = $data['status'] === 'submitted' ? 'submission_status' : 'attendance_status';

        DB::table('exam_session_applicant')
            ->where('exam_session_id', $exam_session->id)
            ->whereIn('applicant_id', $data['applicant_ids'])
            ->update([$column => $data['status']]);

        return back()->with('success', 'Bulk update applied.');
    }
